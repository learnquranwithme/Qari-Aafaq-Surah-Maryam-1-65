١ كٓهٰیٰعٓصٓ
٢ ذِكْرُ رَحْمَتِ رَبِّكَ عَبْدَهٗ زَكَرِیَّا
٣ اِذْ نَادٰی رَبَّهٗ نِدَآءً خَفِیًّا
٤ قَالَ رَبِّ اِنِّیْ وَهَنَ الْعَظْمُ مِنِّیْ وَاشْتَعَلَ الرَّاْسُ شَیْبًا وَّلَمْ اَكُنْ بِدُعَآىِٕكَ رَبِّ شَقِیًّا
٥ وَاِنِّیْ خِفْتُ الْمَوَالِیَ مِنْ وَّرَآءِیْ وَكَانَتِ امْرَاَتِیْ عَاقِرًا فَهَبْ لِیْ مِنْ لَّدُنْكَ وَلِیًّا
٦ یَّرِثُنِیْ وَیَرِثُ مِنْ اٰلِ یَعْقُوْبَ وَاجْعَلْهُ رَبِّ رَضِیًّا
٧ یٰزَكَرِیَّاۤ اِنَّا نُبَشِّرُكَ بِغُلٰمِ سْمُهٗ یَحْیٰی لَمْ نَجْعَلْ لَّهٗ مِنْ قَبْلُ سَمِیًّا
٨ قَالَ رَبِّ اَنّٰی یَكُوْنُ لِیْ غُلٰمٌ وَّكَانَتِ امْرَاَتِیْ عَاقِرًا وَّقَدْ بَلَغْتُ مِنَ الْكِبَرِ عِتِیًّا
٩ قَالَ كَذٰلِكَ قَالَ رَبُّكَ هُوَ عَلَیَّ هَیِّنٌ وَّقَدْ خَلَقْتُكَ مِنْ قَبْلُ وَلَمْ تَكُ شَیْـًٔا
١٠ قَالَ رَبِّ اجْعَلْ لِّیْۤ اٰیَةً قَالَ اٰیَتُكَ اَلَّا تُكَلِّمَ النَّاسَ ثَلٰثَ لَیَالٍ سَوِیًّا
١١ فَخَرَجَ عَلٰی قَوْمِهٖ مِنَ الْمِحْرَابِ فَاَوْحٰۤی اِلَیْهِمْ اَنْ سَبِّحُوْا بُكْرَةً وَّعَشِیًّا
١٢ یٰیَحْیٰی خُذِ الْكِتٰبَ بِقُوَّةٍ وَاٰتَیْنٰهُ الْحُكْمَ صَبِیًّا
١٣ وَّحَنَانًا مِّنْ لَّدُنَّا وَزَكٰوةً وَكَانَ تَقِیًّا
١٤ وَّبَرًّا بِوَالِدَیْهِ وَلَمْ یَكُنْ جَبَّارًا عَصِیًّا
١٥ وَسَلٰمٌ عَلَیْهِ یَوْمَ وُلِدَ وَیَوْمَ یَمُوْتُ وَیَوْمَ یُبْعَثُ حَیًّا
١٦ وَاذْكُرْ فِی الْكِتٰبِ مَرْیَمَ اِذِ انْتَبَذَتْ مِنْ اَهْلِهَا مَكَانًا شَرْقِیًّا
١٧ فَاتَّخَذَتْ مِنْ دُوْنِهِمْ حِجَابًا فَاَرْسَلْنَاۤ اِلَیْهَا رُوْحَنَا فَتَمَثَّلَ لَهَا بَشَرًا سَوِیًّا
١٨ قَالَتْ اِنِّیْۤ اَعُوْذُ بِالرَّحْمٰنِ مِنْكَ اِنْ كُنْتَ تَقِیًّا
١٩ قَالَ اِنَّمَاۤ اَنَا رَسُوْلُ رَبِّكِ لِاَهَبَ لَكِ غُلٰمًا زَكِیًّا
٢٠ قَالَتْ اَنّٰی یَكُوْنُ لِیْ غُلٰمٌ وَّلَمْ یَمْسَسْنِیْ بَشَرٌ وَّلَمْ اَكُ بَغِیًّا
٢١ قَالَ كَذٰلِكِ قَالَ رَبُّكِ هُوَ عَلَیَّ هَیِّنٌ وَلِنَجْعَلَهٗۤ اٰیَةً لِّلنَّاسِ وَرَحْمَةً مِّنَّا وَكَانَ اَمْرًا مَّقْضِیًّا
٢٢ فَحَمَلَتْهُ فَانْتَبَذَتْ بِهٖ مَكَانًا قَصِیًّا
٢٣ فَاَجَآءَهَا الْمَخَاضُ اِلٰی جِذْعِ النَّخْلَةِ قَالَتْ یٰلَیْتَنِیْ مِتُّ قَبْلَ هٰذَا وَكُنْتُ نَسْیًا مَّنْسِیًّا
٢٤ فَنَادٰىهَا مِنْ تَحْتِهَاۤ اَلَّا تَحْزَنِیْ قَدْ جَعَلَ رَبُّكِ تَحْتَكِ سَرِیًّا
٢٥ وَهُزِّیْۤ اِلَیْكِ بِجِذْعِ النَّخْلَةِ تُسٰقِطْ عَلَیْكِ رُطَبًا جَنِیًّا
٢٦ فَكُلِیْ وَاشْرَبِیْ وَقَرِّیْ عَیْنًا فَاِمَّا تَرَیِنَّ مِنَ الْبَشَرِ اَحَدًا فَقُوْلِیْۤ اِنِّیْ نَذَرْتُ لِلرَّحْمٰنِ صَوْمًا فَلَنْ اُكَلِّمَ الْیَوْمَ اِنْسِیًّا
٢٧ فَاَتَتْ بِهٖ قَوْمَهَا تَحْمِلُهٗ قَالُوْا یٰمَرْیَمُ لَقَدْ جِئْتِ شَیْـًٔا فَرِیًّا
٢٨ یٰۤاُخْتَ هٰرُوْنَ مَا كَانَ اَبُوْكِ امْرَاَ سَوْءٍ وَّمَا كَانَتْ اُمُّكِ بَغِیًّا
٢٩ فَاَشَارَتْ اِلَیْهِ قَالُوْا كَیْفَ نُكَلِّمُ مَنْ كَانَ فِی الْمَهْدِ صَبِیًّا
٣٠ قَالَ اِنِّیْ عَبْدُ اللّٰهِ اٰتٰىنِیَ الْكِتٰبَ وَجَعَلَنِیْ نَبِیًّا
٣١ وَّجَعَلَنِیْ مُبٰرَكًا اَیْنَ مَا كُنْتُ وَاَوْصٰنِیْ بِالصَّلٰوةِ وَالزَّكٰوةِ مَا دُمْتُ حَیًّا
٣٢ وَّبَرًّا بِوَالِدَتِیْ وَلَمْ یَجْعَلْنِیْ جَبَّارًا شَقِیًّا
٣٣ وَالسَّلٰمُ عَلَیَّ یَوْمَ وُلِدْتُّ وَیَوْمَ اَمُوْتُ وَیَوْمَ اُبْعَثُ حَیًّا
٣٤ ذٰلِكَ عِیْسَی ابْنُ مَرْیَمَ قَوْلَ الْحَقِّ الَّذِیْ فِیْهِ یَمْتَرُوْنَ
٣٥ مَا كَانَ لِلّٰهِ اَنْ یَّتَّخِذَ مِنْ وَّلَدٍ سُبْحٰنَهٗ اِذَا قَضٰۤی اَمْرًا فَاِنَّمَا یَقُوْلُ لَهٗ كُنْ فَیَكُوْنُ
٣٦ وَاِنَّ اللّٰهَ رَبِّیْ وَرَبُّكُمْ فَاعْبُدُوْهُ هٰذَا صِرَاطٌ مُّسْتَقِیْمٌ
٣٧ فَاخْتَلَفَ الْاَحْزَابُ مِنْ بَیْنِهِمْ فَوَیْلٌ لِّلَّذِیْنَ كَفَرُوْا مِنْ مَّشْهَدِ یَوْمٍ عَظِیْمٍ
٣٨ اَسْمِعْ بِهِمْ وَاَبْصِرْ یَوْمَ یَاْتُوْنَنَا لٰكِنِ الظّٰلِمُوْنَ الْیَوْمَ فِیْ ضَلٰلٍ مُّبِیْنٍ
٣٩ وَاَنْذِرْهُمْ یَوْمَ الْحَسْرَةِ اِذْ قُضِیَ الْاَمْرُ وَهُمْ فِیْ غَفْلَةٍ وَّهُمْ لَا یُؤْمِنُوْنَ
٤٠ اِنَّا نَحْنُ نَرِثُ الْاَرْضَ وَمَنْ عَلَیْهَا وَاِلَیْنَا یُرْجَعُوْنَ
٤١ وَاذْكُرْ فِی الْكِتٰبِ اِبْرٰهِیْمَ اِنَّهٗ كَانَ صِدِّیْقًا نَّبِیًّا
٤٢ اِذْ قَالَ لِاَبِیْهِ یٰۤاَبَتِ لِمَ تَعْبُدُ مَا لَا یَسْمَعُ وَلَا یُبْصِرُ وَلَا یُغْنِیْ عَنْكَ شَیْـًٔا
٤٣ یٰۤاَبَتِ اِنِّیْ قَدْ جَآءَنِیْ مِنَ الْعِلْمِ مَا لَمْ یَاْتِكَ فَاتَّبِعْنِیْۤ اَهْدِكَ صِرَاطًا سَوِیًّا
٤٤ یٰۤاَبَتِ لَا تَعْبُدِ الشَّیْطٰنَ اِنَّ الشَّیْطٰنَ كَانَ لِلرَّحْمٰنِ عَصِیًّا
٤٥ یٰۤاَبَتِ اِنِّیْۤ اَخَافُ اَنْ یَّمَسَّكَ عَذَابٌ مِّنَ الرَّحْمٰنِ فَتَكُوْنَ لِلشَّیْطٰنِ وَلِیًّا
٤٦ قَالَ اَرَاغِبٌ اَنْتَ عَنْ اٰلِهَتِیْ یٰۤاِبْرٰهِیْمُ لَىِٕنْ لَّمْ تَنْتَهِ لَاَرْجُمَنَّكَ وَاهْجُرْنِیْ مَلِیًّا
٤٧ قَالَ سَلٰمٌ عَلَیْكَ سَاَسْتَغْفِرُ لَكَ رَبِّیْ اِنَّهٗ كَانَ بِیْ حَفِیًّا
٤٨ وَاَعْتَزِلُكُمْ وَمَا تَدْعُوْنَ مِنْ دُوْنِ اللّٰهِ وَاَدْعُوْا رَبِّیْ عَسٰۤی اَلَّاۤ اَكُوْنَ بِدُعَآءِ رَبِّیْ شَقِیًّا
٤٩ فَلَمَّا اعْتَزَلَهُمْ وَمَا یَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ وَهَبْنَا لَهٗۤ اِسْحٰقَ وَیَعْقُوْبَ وَكُلًّا جَعَلْنَا نَبِیًّا
٥٠ وَوَهَبْنَا لَهُمْ مِّنْ رَّحْمَتِنَا وَجَعَلْنَا لَهُمْ لِسَانَ صِدْقٍ عَلِیًّا
٥١ وَاذْكُرْ فِی الْكِتٰبِ مُوْسٰۤی اِنَّهٗ كَانَ مُخْلَصًا وَّكَانَ رَسُوْلًا نَّبِیًّا
٥٢ وَنَادَیْنٰهُ مِنْ جَانِبِ الطُّوْرِ الْاَیْمَنِ وَقَرَّبْنٰهُ نَجِیًّا
٥٣ وَوَهَبْنَا لَهٗ مِنْ رَّحْمَتِنَاۤ اَخَاهُ هٰرُوْنَ نَبِیًّا
٥٤ وَاذْكُرْ فِی الْكِتٰبِ اِسْمٰعِیْلَ اِنَّهٗ كَانَ صَادِقَ الْوَعْدِ وَكَانَ رَسُوْلًا نَّبِیًّا
٥٥ وَكَانَ یَاْمُرُ اَهْلَهٗ بِالصَّلٰوةِ وَالزَّكٰوةِ وَكَانَ عِنْدَ رَبِّهٖ مَرْضِیًّا
٥٦ وَاذْكُرْ فِی الْكِتٰبِ اِدْرِیْسَ اِنَّهٗ كَانَ صِدِّیْقًا نَّبِیًّا
٥٧ وَّرَفَعْنٰهُ مَكَانًا عَلِیًّا
٥٨ اُولٰٓىِٕكَ الَّذِیْنَ اَنْعَمَ اللّٰهُ عَلَیْهِمْ مِّنَ النَّبِیّٖنَ مِنْ ذُرِّیَّةِ اٰدَمَ وَمِمَّنْ حَمَلْنَا مَعَ نُوْحٍ وَّمِنْ ذُرِّیَّةِ اِبْرٰهِیْمَ وَاِسْرَآءِیْلَ وَمِمَّنْ هَدَیْنَا وَاجْتَبَیْنَا اِذَا تُتْلٰی عَلَیْهِمْ اٰیٰتُ الرَّحْمٰنِ خَرُّوْا سُجَّدًا وَّبُكِیًّا
٥٩ فَخَلَفَ مِنْ بَعْدِهِمْ خَلْفٌ اَضَاعُوا الصَّلٰوةَ وَاتَّبَعُوا الشَّهَوٰتِ فَسَوْفَ یَلْقَوْنَ غَیًّا
٦٠ اِلَّا مَنْ تَابَ وَاٰمَنَ وَعَمِلَ صَالِحًا فَاُولٰٓىِٕكَ یَدْخُلُوْنَ الْجَنَّةَ وَلَا یُظْلَمُوْنَ شَیْـًٔا
٦١ جَنّٰتِ عَدْنِ لَّتِیْ وَعَدَ الرَّحْمٰنُ عِبَادَهٗ بِالْغَیْبِ اِنَّهٗ كَانَ وَعْدُهٗ مَاْتِیًّا
٦٢ لَا یَسْمَعُوْنَ فِیْهَا لَغْوًا اِلَّا سَلٰمًا وَلَهُمْ رِزْقُهُمْ فِیْهَا بُكْرَةً وَّعَشِیًّا
٦٣ تِلْكَ الْجَنَّةُ الَّتِیْ نُوْرِثُ مِنْ عِبَادِنَا مَنْ كَانَ تَقِیًّا
٦٤ وَمَا نَتَنَزَّلُ اِلَّا بِاَمْرِ رَبِّكَ لَهٗ مَا بَیْنَ اَیْدِیْنَا وَمَا خَلْفَنَا وَمَا بَیْنَ ذٰلِكَ وَمَا كَانَ رَبُّكَ نَسِیًّا
٦٥ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَیْنَهُمَا فَاعْبُدْهُ وَاصْطَبِرْ لِعِبَادَتِهٖ هَلْ تَعْلَمُ لَهٗ سَمِیًّا
